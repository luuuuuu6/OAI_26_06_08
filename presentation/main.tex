\begin{frame}{Adaptive Filtering — Three Algorithms}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.32\textwidth}
\begin{block}{EWMA}
\begin{itemize}
  \item Fixed $\alpha$
  \item Simplest baseline
  \item Sensitive to outlier frames
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.32\textwidth}
\begin{block}{IBVSS}
\begin{itemize}
  \item $\alpha = 1 - 1/\text{ratio}$
  \item Kalman steady-state mapping
  \item Zero calibration needed
  \item \textbf{Most robust in practice}
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.32\textwidth}
\begin{block}{Scalar Kalman}
\begin{itemize}
  \item Full Riccati recursion
  \item IAE online $Q$ estimation
  \item Best at high SNR
  \item Slower warmup (30 frames)
\end{itemize}
\end{block}
\end{column}
\end{columns}

\vspace{4pt}
\callout{\small Key improvements applied to IBVSS and Kalman: P4 \textit{per-band median} innovation ($B{=}8$) + bilateral $d$ normalization}
\end{frame}

\begin{block}{Compressed $H$}
\small
\begin{itemize}
  \item \texttt{h\_taps}: delay-domain kept taps, shape $(2{\cdot}N_\text{tap})$
  \item \texttt{H\_c}: full-band FFT reconstruction, shape $(N_\text{sc})$
\end{itemize}
\end{block}

\vspace{-2pt}
\begin{block}{Update rules}
\scriptsize
\textbf{EMA convention}
\[
  x_\text{state}[t] = (1{-}\alpha)\,x_\text{state}[t{-}1] + \alpha\,x_\text{inst}[t]
\]
\vspace{-9pt}
\textbf{Doppler autocorrelation}
\[
  d[\tau] = \frac{|\langle H_t, H_{t{-}\tau}\rangle|}
                 {\sqrt{\|H_t\|^2 \cdot \|H_{t{-}\tau}\|^2}}
\]
\end{block}
\end{column}
\end{columns}
\end{frame}

\textbf{Algorithm} & \textbf{5\,dB} & \textbf{10\,dB} & \textbf{15\,dB} & \textbf{20\,dB} \\

\multicolumn{5}{c}{\textit{NMSE $P_{50}$ (dB) — median, lower is better}} \\
\midrule
EWMA   & $-9.33$  & $-0.62$\textsuperscript{*}  & $-9.27$  & $-8.88$ \\
IBVSS  & $\mathbf{-10.46}$ & $\mathbf{-10.06}$ & $\mathbf{-10.87}$ & $-11.00$ \\
Kalman & $+21.52$\textsuperscript{$\dagger$} & $-11.19$ & $-9.92$ & $\mathbf{-11.45}$ \\
\midrule
\multicolumn{5}{c}{\textit{NMSE $P_{90}$ (dB) — worst 10\%, lower is better}} \\
\midrule
EWMA   & $+25.87$ & $+25.40$ & $-1.72$  & $+1.65$ \\
IBVSS  & $\mathbf{-7.77}$  & $\mathbf{-8.55}$  & $\mathbf{-6.41}$  & $+21.55$\textsuperscript{$\ddagger$} \\
Kalman & $+27.00$ & $-7.00$  & $+25.25$ & $\mathbf{-8.58}$ \\
\bottomrule
\end{tabular}
\end{adjustbox}

\vspace{4pt}
\scriptsize
\textsuperscript{*}EWMA 10\,dB: data quality issue. \quad
\textsuperscript{$\dagger$}Kalman 5\,dB: warmup disaster. \quad
\textsuperscript{$\ddagger$}IBVSS 20\,dB $P_{90}$: timing-glitch outlier.
\end{frame}

\begin{frame}{Key Findings}
\small
\begin{enumerate}
  \item \textbf{IBVSS is the most robust algorithm} \\
        $P_{50} \in [-10, -11]$\,dB across all SNR — no catastrophic failure
  \item \textbf{Kalman is optimal at high SNR} \\
        $-11.45$\,dB at 20\,dB, but warmup sensitivity at low SNR
  \item \textbf{Adaptive $\alpha$ suppresses worst-case frames} \\
        IBVSS $P_{90}$: $[-6, -9]$\,dB \quad vs \quad EWMA: $[+1, +26]$\,dB
\end{enumerate}

\vspace{4pt}
\callout{\small For vRAN deployment, consistent $-10$\,dB is better than ``sometimes $-11.5$, sometimes $+21.5$''.}
\end{frame}

\vspace{5pt}
\raggedright
\scriptsize
\textbf{PASS criterion} (Capture Gate, $\varepsilon{=}14\%$): \\[2pt]
Each condition = (SNR, speed) pair, 5--20 trials averaged. \\
PASS $\Leftrightarrow$ $\text{MSE}_\text{adaptive} \;\le\; (1{+}\varepsilon) \times \text{MSE}_\text{oracle}$,
where oracle = best \textit{fixed} $\alpha$ found by grid search. \\[2pt]
$n/m$ = conditions passed / total conditions tested. \\
CDL-C/D/A: 8 conditions = \{0, 5, 10, 20\}\,dB $\times$ \{3, 30\}\,km/h. \quad
P1B: 6 conditions = \{0, 10, 20\}\,dB $\times$ \{3, 30\}\,km/h.
